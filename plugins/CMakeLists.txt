# plugins/ — first-party plugins for the LightC++ IDE.
#
# Shared-library plugins are QPluginLoader DLLs loaded once at start-up from
# <exe dir>/plugins. Lua scripts under lua/ are loaded by the app's sandboxed
# Lua host from <exe dir>/plugins/lua.

# The language/framework backends below are pure Qt Core code: they only
# parse a descriptor and locate the host tool on PATH. They build on every
# platform the IDE supports, unlike the Windows-only Dev-C++/MSVC backends.
function(lightcpp_add_project_plugin target source)
    add_library(${target} MODULE
        ${source}
        metadata.json
    )
    set_target_properties(${target} PROPERTIES
        AUTOMOC ON
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/plugins
        LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/plugins
    )
    target_include_directories(${target} PRIVATE
        ${CMAKE_SOURCE_DIR}/src/LightCppPlugin/Include
        ${CMAKE_SOURCE_DIR}/plugins/Support
    )
    target_link_libraries(${target} PRIVATE
        LightCppPlugin
        Qt6::Core
        LightCppWarnings
    )
    if(MINGW)
        # Same runtime model as the IDE executable: no toolchain DLLs needed.
        target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
    if(TARGET lightcpp)
        add_dependencies(lightcpp ${target})
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory
                    ${CMAKE_BINARY_DIR}/dist/plugins
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    $<TARGET_FILE:${target}>
                    ${CMAKE_BINARY_DIR}/dist/plugins/
            COMMENT "Staging ${target} into dist/plugins"
        )
    endif()
endfunction()

# Dev-C++ makefiles generated by this backend use cmd.exe, windres.exe
# and Windows resource/linker conventions. Do not advertise that backend on
# POSIX until it has a native makefile implementation.
if(WIN32)
    add_subdirectory(DevCppPlugin)
    # MSBuild, vswhere and the Windows SDK registry are Windows-specific.
    add_subdirectory(MsvcPlugin)
endif()

add_subdirectory(GoPlugin)

add_subdirectory(QtCppPlugin)
